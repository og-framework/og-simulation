cmake_minimum_required(VERSION 3.20)
project(OGTickDriver CXX)

if(PROJECT_IS_TOP_LEVEL)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    if(MSVC)
        add_compile_options(/Zc:preprocessor /utf-8)
    endif()
    add_compile_definitions(OG_STANDALONE_BUILD)
endif()

if(NOT TARGET og_simulation)
    set(OG_SIMULATION_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Root of an og-simulation checkout (the directory that holds OGSimulation/ and glm/)")
    if(NOT EXISTS "${OG_SIMULATION_DIR}/OGSimulation/CMakeLists.txt")
        message(FATAL_ERROR "og-tickdriver: set OG_SIMULATION_DIR to the root of an og-simulation checkout")
    endif()
    if(NOT TARGET glm::glm-header-only)
        add_subdirectory("${OG_SIMULATION_DIR}/glm" "${CMAKE_CURRENT_BINARY_DIR}/glm")
    endif()
    add_subdirectory("${OG_SIMULATION_DIR}/OGSimulation" "${CMAKE_CURRENT_BINARY_DIR}/og_simulation")
endif()

file(GLOB OGTICKDRIVER_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/OGTickDriver/*.cpp")
file(GLOB OGTICKDRIVER_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/OGTickDriver/*.h")

add_library(og_tickdriver STATIC ${OGTICKDRIVER_SOURCES} ${OGTICKDRIVER_HEADERS})
target_include_directories(og_tickdriver PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(og_tickdriver PUBLIC cxx_std_20)
target_link_libraries(og_tickdriver PUBLIC og_simulation)
if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    target_compile_options(og_tickdriver PUBLIC /fp:precise)
elseif(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(og_tickdriver PUBLIC /fp:precise /clang:-ffp-contract=off)
else()
    target_compile_options(og_tickdriver PUBLIC -ffp-contract=off)
endif()

set(OGTICKDRIVER_HEADER_TUS "")
foreach(header IN LISTS OGTICKDRIVER_HEADERS)
    file(RELATIVE_PATH header_rel "${CMAKE_CURRENT_SOURCE_DIR}" "${header}")
    string(MAKE_C_IDENTIFIER "${header_rel}" header_id)
    set(header_tu "${CMAKE_CURRENT_BINARY_DIR}/header_check/${header_id}.cpp")
    set(header_tu_text "#include \"${header_rel}\"\n")
    if(EXISTS "${header_tu}")
        file(READ "${header_tu}" header_tu_old)
    else()
        set(header_tu_old "")
    endif()
    if(NOT header_tu_old STREQUAL header_tu_text)
        file(WRITE "${header_tu}" "${header_tu_text}")
    endif()
    list(APPEND OGTICKDRIVER_HEADER_TUS "${header_tu}")
endforeach()

add_library(og_tickdriver_header_check OBJECT ${OGTICKDRIVER_HEADER_TUS})
target_link_libraries(og_tickdriver_header_check PRIVATE og_tickdriver)
if(MSVC)
    target_compile_options(og_tickdriver_header_check PRIVATE /GR- /EHs-c- /wd4577)
    target_compile_definitions(og_tickdriver_header_check PRIVATE _HAS_EXCEPTIONS=0)
else()
    target_compile_options(og_tickdriver_header_check PRIVATE -fno-rtti -fno-exceptions)
endif()
